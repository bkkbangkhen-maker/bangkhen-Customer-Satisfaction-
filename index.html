โค้ดสำหรับเชื่อมต่อ Google Sheets (Apps Script)

1. สร้างไฟล์ Google Sheets ใหม่ ตั้งชื่อชีตด้านล่างว่า Responses

2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script ลบโค้ดเดิมออกแล้ววางโค้ดด้านล่างนี้:

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Responses');
  if (!sheet) {
    sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet('Responses');
    sheet.appendRow(['ID', 'Timestamp', 'วันที่', 'ฝ่ายงาน', 'ช่องบริการ', 'คะแนน', 'ระดับความพึงพอใจ', 'ข้อเสนอแนะ', 'ช่างรังวัด']);
  }
  var data = JSON.parse(e.postData.contents);
  var scoreLabels = {'5':'พอใจมากที่สุด','4':'พอใจมาก','3':'พอใจปานกลาง','2':'พอใจน้อย','1':'พอใจน้อยที่สุด'};
  sheet.appendRow([
    data.id,
    data.timestamp,
    data.date,
    data.department,
    data.counter,
    data.score,
    scoreLabels[String(data.score)] || '',
    data.suggestion || '',
    data.surveyor || ''
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Responses');
  if (!sheet) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (let i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    let dateVal = rows[i][2];
    if (dateVal instanceof Date) {
      dateVal = Utilities.formatDate(dateVal, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    }
    result.push({
      id: String(rows[i][0]),
      timestamp: String(rows[i][1]),
      date: String(dateVal),
      department: String(rows[i][3]),
      counter: Number(rows[i][4]),
      score: Number(rows[i][5]),
      suggestion: String(rows[i][7] || ''),
      surveyor: String(rows[i][8] || '')
    });
  }
  return ContentService.createTextOutput(JSON.stringify(result.reverse())).setMimeType(ContentService.MimeType.JSON);
}
